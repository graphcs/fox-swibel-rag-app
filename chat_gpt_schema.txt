              type: object

                  type: string
                  description: The unique identifier for the specific document (optional).
                clientId:
                  type: string
                  description: The client identifier associated with the matter.
                operation:
                  type: string

                    - summarize
                    - analyze
                    - list_files
                  description: The specific action to perform on the target data.
                userEmail:
                  type: string
                  description: The user's email address
              required:

                - operation
      responses:

          description: Successful operation
          content:
            application/json:
              schema:
                type: object

                  status:
                    type: string
                  data:
                    type: string
